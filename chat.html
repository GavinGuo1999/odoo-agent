SELECT
    date_trunc('month', date_order) AS month,
    SUM(amount_untaxed) AS sales_amount,
    COUNT(*) AS order_count
FROM sale_order
WHERE state IN ('sale', 'done')
  AND date_order >= date_trunc('month', CURRENT_DATE) - INTERVAL '1 month'
GROUP BY 1
ORDER BY 1;
